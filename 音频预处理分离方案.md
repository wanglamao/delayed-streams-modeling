# Kyutai STT 音频预处理分离方案

将 Mimi 编码从训练循环中剥离，预生成二进制 records，加速训练。

---

## 1. 设计

### 1.1 核心思路

```
预处理（一次性）:  WAV + alignments → Mimi encode → safetensors shards
训练:              load shards → Transformer 训练
```

**收益**: 无需重复 Mimi encode，训练加速 30-50%，显存节省 ~500MB。

### 1.2 分片策略

```
data/
  train/
    shard_00000.safetensors  (10k samples)
    shard_00001.safetensors
    ...
    meta.json                # 索引信息
```

**单个 shard 结构** (safetensors 格式):
```python
{
    # 共享维度信息
    "num_samples": tensor([N]),           # [1], int64
    "num_frames":  tensor([T]),           # [1], int64, frames per sample

    # 实际数据 (flattened)
    "codes": tensor([N, 33, T], dtype=int16),  # all codes in this shard
    "masks": tensor([N, T], dtype=bool),       # valid frames mask (for padding)
}

# Metadata (json header)
{
    "paths": ["path/to/audio1.wav", "path/to/audio2.wav", ...],
    "start_secs": [0.0, 10.0, ...],
}
```

**为什么要 flatten**:
- 每个 sample 长度相同（固定 duration），可以直接 stack
- safetensors 对单一连续 tensor 的 mmap 效率最高
- 无需逐个 sample 读取

### 1.3 索引文件

```json
// meta.json
{
    "num_shards": 10,
    "total_samples": 100000,
    "samples_per_shard": 10000,
    "config_hash": "a3f2...8c9d",
    "shards": [
        {"path": "shard_00000.safetensors", "num_samples": 10000},
        {"path": "shard_00001.safetensors", "num_samples": 10000},
        ...
    ]
}
```

### 2.2 预处理脚本

```python
# scripts/preprocess_stt_dataset.py
import json
import torch
from pathlib import Path
from safetensors.torch import save_file


def preprocess_shard(args, shard_id: int, samples: list):
    """处理单个 shard"""
    all_codes = []
    all_masks = []
    metadata = {"paths": [], "start_secs": []}

    for wav_path, start_sec in samples:
        # 1. Load & Mimi encode
        wav = load_wav(wav_path, start_sec, duration_sec=20)
        audio_tokens = mimi.encode(wav)  # [1, 32, T]

        # 2. Build text tokens from alignments
        text_tokens = interleaver.prepare_item(alignments, T)

        # 3. Concat
        codes = torch.cat([text_tokens, audio_tokens], dim=1)  # [1, 33, T]

        # 4. Pad to fixed length (if needed)
        mask = torch.ones(T, dtype=torch.bool)
        if codes.shape[-1] < T:
            pad_len = T - codes.shape[-1]
            codes = torch.nn.functional.pad(codes, (0, pad_len), value=0)
            mask[codes.shape[-1]:] = False

        all_codes.append(codes[0].to(torch.int16))  # [33, T]
        all_masks.append(mask)
        metadata["paths"].append(str(wav_path))
        metadata["start_secs"].append(start_sec)

    # 5. Stack and save
    shard_data = {
        "codes": torch.stack(all_codes),  # [N, 33, T], int16
        "masks": torch.stack(all_masks),  # [N, T], bool
        "num_samples": torch.tensor([len(all_codes)], dtype=torch.int64),
        "num_frames": torch.tensor([T], dtype=torch.int64),
    }

    output_path = Path(args.output_dir) / f"shard_{shard_id:05d}.safetensors"
    save_file(shard_data, output_path, metadata=metadata)

    return output_path, len(all_codes)


def main():
    # 分片处理
    samples = load_jsonl_samples(args.input)
    shards = chunk(samples, args.samples_per_shard)

    shard_infos = []
    for shard_id, shard_samples in enumerate(shards):
        path, num = preprocess_shard(args, shard_id, shard_samples)
        shard_infos.append({"path": str(path), "num_samples": num})

    # 写 meta.json
    meta = {
        "num_shards": len(shards),
        "total_samples": len(samples),
        "samples_per_shard": args.samples_per_shard,
        "config_hash": compute_config_hash(args),
        "shards": shard_infos,
    }
    json.dump(meta, open(Path(args.output_dir) / "meta.json", "w"))
```

### 2.3 DataLoader 适配

```python
# finetune/data/records_dataset.py
import json
import torch
from pathlib import Path
from torch.utils.data import Dataset, Sampler
from safetensors.torch import load_file


class SafetensorsShardDataset(Dataset):
    """单个 shard，支持 mmap"""

    def __init__(self, shard_path: str):
        self.shard_path = Path(shard_path)
        # mmap 加载，零拷贝
        self.data = load_file(self.shard_path, device="cpu")
        self.num_samples = int(self.data["num_samples"].item())

    def __len__(self):
        return self.num_samples

    def __getitem__(self, idx: int):
        codes = self.data["codes"][idx].long()  # [33, T], 转回 int64
        mask = self.data["masks"][idx]          # [T]
        return {"codes": codes, "mask": mask}


class ShardedRecordsDataset(Dataset):
    """多 shard 数据集"""

    def __init__(self, records_dir: str):
        self.records_dir = Path(records_dir)
        self.meta = json.load(open(self.records_dir / "meta.json"))

        # 延迟加载 shards（按需 mmap）
        self.shards = []
        self.shard_offsets = [0]
        for info in self.meta["shards"]:
            shard = SafetensorsShardDataset(self.records_dir / info["path"])
            self.shards.append(shard)
            self.shard_offsets.append(self.shard_offsets[-1] + len(shard))

        # 验证 config hash
        # current_hash = compute_config_hash(...)
        # assert current_hash == self.meta["config_hash"]

    def __len__(self):
        return self.shard_offsets[-1]

    def __getitem__(self, idx: int):
        # 找到对应的 shard
        shard_idx = bisect_right(self.shard_offsets, idx) - 1
        local_idx = idx - self.shard_offsets[shard_idx]
        return self.shards[shard_idx][local_idx]


# Collate function（保持不变）
def collate_fn(batch):
    codes = torch.stack([b["codes"] for b in batch])  # [B, 33, T]
    masks = torch.stack([b["mask"] for b in batch])   # [B, T]
    return Batch(codes=codes, masks=masks)
```

## 4. 对比：自定义格式 vs safetensors

| 特性 | 自定义二进制 | safetensors |
|------|-------------|-------------|
| 实现成本 | 高（需自己写序列化） | 低（现成 API） |
| mmap 支持 | 需自己实现 | 内置 |
| 校验和 | 需自己实现 | 内置 CRC32 |
| 安全性 | 一般 | 安全（防止 pickle 漏洞） |
| 跨平台 | 需注意字节序 | 自动处理 |
| 依赖 | 无 | safetensors（已有） |

**结论**：使用 safetensors，简单可靠。
